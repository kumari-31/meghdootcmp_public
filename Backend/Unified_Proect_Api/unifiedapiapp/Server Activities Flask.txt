#!/bin/bash

# Declare an array to store VM names
declare -a vms

# Collect command-line arguments (VM names)
vms=("$@")

# Get the directory of the script
script_dir=$(dirname "$(readlink -f "$0")")
csv_file="$script_dir/vm_info_new.csv"

# Source the OpenStack credentials
source /home/cloud/openrc

# Create a CSV file with a header
echo "VMname,Hostname,Instance Name,VNC Display" > "$csv_file"

# Loop through each VM in the list
for vm in "${vms[@]}"; do
    # Get the host information
    host=$(openstack server show "$vm" | awk '/OS-EXT-SRV-ATTR:host/ {print $4}')
    #echo "$host"
    # Get the instance name information
    instance=$(openstack server show "$vm" | awk '/OS-EXT-SRV-ATTR:instance_name/ {print $4}')
    #echo  "$instance"
    # Check if the host is the local server
    if [ "$host" == "meghnode" ]; then
        # Get the VNC display for the instance
        vnc=$(sudo virsh vncdisplay "$instance" | awk -F: '{print $2}')
	#echo "$vnc"        
        # Write the data to the CSV file
        echo "$vm,$host,$instance,$vnc" >> "$csv_file"
    fi
done

echo "VM information saved to $csv_file"


server.py

import subprocess
from flask import Flask, request, jsonify
import os

app = Flask(__name__)

# Path to your Bash script, using the current directory of this script
SCRIPT_PATH = os.path.join(os.path.dirname(os.path.realpath(__file__)), "fetch_vm_info.sh")

@app.route('/fetch-vm-info', methods=['POST'])
def fetch_vm_info():
    # Get the VM names from the POST request JSON body
    data = request.json
    if not data or "vms" not in data:
        return jsonify({"error": "No VMs provided"}), 400
    
    vm_names = data["vms"]

    # Ensure VM names are provided as a list
    if not isinstance(vm_names, list) or not vm_names:
        return jsonify({"error": "VM names must be a non-empty list"}), 400
    
    try:
        # Run the Bash script with the VM names as arguments
        result = subprocess.run(
            [SCRIPT_PATH, *vm_names],  # Pass VM names to the script
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True
        )
        
        # Check if the script ran successfully
        if result.returncode != 0:
            return jsonify({"error": "Script execution failed", "details": result.stderr}), 500
        
        # Parse the output CSV file generated by the script
        output_file = os.path.join(os.path.dirname(SCRIPT_PATH), "vm_info_new.csv")
        if not os.path.exists(output_file):
            return jsonify({"error": "Output CSV file not found"}), 500
        
        # Read and return the CSV content
        with open(output_file, "r") as f:
            csv_content = f.read()
        
        return jsonify({
            "message": "Script executed successfully",
            "output_csv": csv_content
        })

    except Exception as e:
        return jsonify({"error": "An error occurred", "details": str(e)}), 500

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8001)
